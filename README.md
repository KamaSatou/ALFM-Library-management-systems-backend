All changes can be made in application.properties.<br>
Change the Oracle plug-in to ORCLPDB and grant the necessary access to that account.<br>
Tabble Setting:
```sql
-- ============================================
-- 1. CATEGORIES
-- ============================================

CREATE TABLE LIB_CATEGORIES (
    CategoryID NUMBER GENERATED BY DEFAULT AS IDENTITY,
    CategoryName VARCHAR2(200) NOT NULL,

    CONSTRAINT PK_LIB_CATEGORIES
        PRIMARY KEY (CategoryID),

    CONSTRAINT UQ_LIB_CATEGORY_NAME
        UNIQUE (CategoryName)
);


-- ============================================
-- 2. AUTHORS
-- ============================================

CREATE TABLE LIB_AUTHORS (
    AuthorID NUMBER GENERATED BY DEFAULT AS IDENTITY,
    AuthorName VARCHAR2(300) NOT NULL,

    CONSTRAINT PK_LIB_AUTHORS
        PRIMARY KEY (AuthorID)
);


-- ============================================
-- 3. BOOKS
-- ============================================

CREATE TABLE LIB_BOOKS (
    BookID NUMBER GENERATED BY DEFAULT AS IDENTITY,
    Title VARCHAR2(500) NOT NULL,
    ISBN VARCHAR2(30),
    Tag VARCHAR2(100),
    Series VARCHAR2(300),

    CategoryID NUMBER,
    AuthorID NUMBER,

    Quantity NUMBER DEFAULT 1 NOT NULL,
    AvailableQuantity NUMBER DEFAULT 1 NOT NULL,

    CONSTRAINT PK_LIB_BOOKS
        PRIMARY KEY (BookID),

    CONSTRAINT FK_LIB_BOOK_CATEGORY
        FOREIGN KEY (CategoryID)
        REFERENCES LIB_CATEGORIES(CategoryID),

    CONSTRAINT FK_LIB_BOOK_AUTHOR
        FOREIGN KEY (AuthorID)
        REFERENCES LIB_AUTHORS(AuthorID),

    CONSTRAINT CK_LIB_BOOK_QUANTITY
        CHECK (Quantity >= 0),

    CONSTRAINT CK_LIB_BOOK_AVAILABLE
        CHECK (AvailableQuantity >= 0),

    CONSTRAINT CK_LIB_BOOK_AVAILABLE_MAX
        CHECK (AvailableQuantity <= Quantity)
);


-- ============================================
-- 4. BOOK IMAGES
-- ============================================

CREATE TABLE LIB_BOOK_IMAGES (
    ImageID NUMBER GENERATED BY DEFAULT AS IDENTITY,
    BookID NUMBER NOT NULL,

    ImageName VARCHAR2(300),
    MaskID NUMBER,
    OCRText CLOB,

    CONSTRAINT PK_LIB_BOOK_IMAGES
        PRIMARY KEY (ImageID),

    CONSTRAINT FK_LIB_IMAGE_BOOK
        FOREIGN KEY (BookID)
        REFERENCES LIB_BOOKS(BookID)
        ON DELETE CASCADE
);


-- ============================================
-- 5. MEMBERS
-- ============================================

CREATE TABLE LIB_MEMBERS (
    MemberID NUMBER GENERATED BY DEFAULT AS IDENTITY,
    FullName VARCHAR2(200) NOT NULL,
    Email VARCHAR2(200),
    Phone VARCHAR2(30),

    RegisterDate DATE DEFAULT SYSDATE NOT NULL,
    Status VARCHAR2(20) DEFAULT 'ACTIVE' NOT NULL,

    CONSTRAINT PK_LIB_MEMBERS
        PRIMARY KEY (MemberID),

    CONSTRAINT CK_LIB_MEMBER_STATUS
        CHECK (Status IN ('ACTIVE', 'INACTIVE'))
);


-- ============================================
-- 6. BORROWS
-- ============================================

CREATE TABLE LIB_BORROWS (
    BorrowID NUMBER GENERATED BY DEFAULT AS IDENTITY,
    MemberID NUMBER NOT NULL,

    BorrowDate DATE DEFAULT SYSDATE NOT NULL,
    DueDate DATE NOT NULL,

    Status VARCHAR2(20) DEFAULT 'BORROWING' NOT NULL,

    CONSTRAINT PK_LIB_BORROWS
        PRIMARY KEY (BorrowID),

    CONSTRAINT FK_LIB_BORROW_MEMBER
        FOREIGN KEY (MemberID)
        REFERENCES LIB_MEMBERS(MemberID),

    CONSTRAINT CK_LIB_BORROW_STATUS
        CHECK (Status IN ('BORROWING', 'RETURNED', 'OVERDUE'))
);


-- ============================================
-- 7. BORROW DETAILS
-- ============================================

CREATE TABLE LIB_BORROW_DETAILS (
    BorrowID NUMBER NOT NULL,
    BookID NUMBER NOT NULL,

    ReturnDate DATE,

    CONSTRAINT PK_LIB_BORROW_DETAILS
        PRIMARY KEY (BorrowID, BookID),

    CONSTRAINT FK_LIB_DETAIL_BORROW
        FOREIGN KEY (BorrowID)
        REFERENCES LIB_BORROWS(BorrowID)
        ON DELETE CASCADE,

    CONSTRAINT FK_LIB_DETAIL_BOOK
        FOREIGN KEY (BookID)
        REFERENCES LIB_BOOKS(BookID)
);
CREATE TABLE ACCOUNTS (
    ACCOUNTID NUMBER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    MEMBERID NUMBER NOT NULL,
    EMAIL VARCHAR2(200 BYTE) NOT NULL,
    PASSWORD VARCHAR2(255 BYTE) NOT NULL,
    ROLE VARCHAR2(30 BYTE) DEFAULT 'MEMBER' NOT NULL,
    STATUS VARCHAR2(20 BYTE) DEFAULT 'ACTIVE' NOT NULL,
    RESETOTP VARCHAR2(20 BYTE),
    RESETOTPEXPIRES TIMESTAMP(6),

    CONSTRAINT PK_ACCOUNTS PRIMARY KEY (ACCOUNTID),

    CONSTRAINT FK_ACCOUNT_MEMBER
        FOREIGN KEY (MEMBERID)
        REFERENCES LIB_MEMBERS (MEMBERID)
        ON DELETE NO ACTION,

    CONSTRAINT UQ_ACCOUNT_EMAIL UNIQUE (EMAIL),

    CONSTRAINT UQ_ACCOUNT_MEMBER UNIQUE (MEMBERID)
);

CREATE SEQUENCE SEQ_LIB_BOOKS
START WITH 9
INCREMENT BY 1
NOCACHE
NOCYCLE;
```
